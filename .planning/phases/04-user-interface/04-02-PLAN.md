**Self-Elevation Check (App.xaml.cs)** — check at startup, relaunch with `runas` if not elevated:
```csharp
private static bool IsElevated()
{
    using var id = WindowsIdentity.GetCurrent();
    return id.Groups.Any(g => g.TValue == "S-1-5-32-544"); // Administrators
}

protected override void OnLaunched(LaunchActivatedEventArgs args)
{
    if (!IsElevated())
    {
        var dialog = new ContentDialog
        {
            Title = "Administrator Required",
            Content = "Akari Tool V2 requires administrator privileges to apply system tweaks. The app will now restart with elevated permissions.",
            CloseButtonText = "OK"
        };
        dialog.ShowAsync().AsTask().Wait(TimeSpan.FromSeconds(30));

        var startInfo = new ProcessStartInfo
        {
            FileName = Assembly.GetExecutingAssembly().Location,
            UseShellExecute = true,
            Verb = "runas"  // triggers UAC prompt
        };
        Process.Start(startInfo);
        Shutdown();
        return;
    }

    // proceed with normal launch...
}
```

### Batch Apply with Progress (UI-05)
**MainViewModel** additions:
```csharp
public ObservableCollection<TweakViewModel> SelectedTweaks { get; }
// bound via XAML: <CheckBox IsChecked="{x:Bind IsSelected, Mode=TwoWay}" />

private double _batchProgress;
public double BatchProgress
{
    get => _batchProgress;
    set => SetProperty(ref _batchProgress, value);
}

private bool _isBatchApplying;
public bool IsBatchApplying
{
    get => _batchApplying;
    set => SetProperty(ref _isBatchApplying, value);
}

public IAsyncRelayCommand ApplySelectedCommand { get; }

// Implementation:
private async Task ApplySelectedAsync()
{
    IsBatchApplying = true;
    BatchProgress = 0;
    
    var selectedIds = SelectedTweaks.Where(t => t.IsSelected).Select(t => t.Id).ToList();
    var results = await _engine.ApplyBatchAsync(selectedIds);
    
    for (int i = 0; i < results.Count; i++)
    {
        // update each TweakViewModel's IsApplied based on result
        var vm = SelectedTweaks.FirstOrDefault(t => t.Id == results[i].TweakId);
        if (vm != null) vm.IsApplied = results[i].Success && results[i].Status == TweakStatus.Applied;
        
        BatchProgress = (double)(i + 1) / results.Count * 100;
    }
    
    IsBatchApplying = false;
}
```

**MainWindow.xaml** additions:
```xml
<!-- Progress bar at bottom of checklist -->
<ProgressBar Value="{x:Bind ViewModel.BatchProgress, Mode=OneWay}"
             IsIndeterminate="{x:Bind ViewModel.IsBatchApplying, Mode=OneWay, Converter={StaticResource InverseBoolConverter}}"
             Height="10" />

<!-- Apply Selected button -->
<Button Content="Apply Selected"
        Command="{x:Bind ViewModel.ApplySelectedCommand}"
        IsEnabled="{x:Bind ViewModel.HasSelectedTweaks, Mode=OneWay}" />
```

## Risks
- **Elevation relaunch**: Assembly.GetExecutingAssembly().Location may resolve to the publish directory — must ensure correct path
- **ContentDialog async**: Must use `.ShowAsync().AsTask().Wait()` pattern in OnLaunched (no pure async in OnLaunched)
- **Batch progress binding**: Need `InverseBoolConverter` to toggle ProgressBar between determinate and indeterminate
- **Race condition**: ApplyBatchAsync is already Task.Run-offloaded; progress updates must be marshaled to UI thread
